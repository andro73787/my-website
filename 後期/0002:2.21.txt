#include <M5StickCPlus2.h>
#include <DHT.h>

// DHT11 Configuration
#define DHTPIN 26
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

// LoRa Pins
#define LORA_RX 33
#define LORA_TX 32


// Variables
unsigned long lastSend = 0;
const unsigned long sendInterval = 10000;  // 10 seconds
int sendCount = 0;
bool dhtReady = false;

void setup() {
    // Initialize M5StickC Plus2
    auto cfg = M5.config();
    StickCP2.begin(cfg);
    
    Serial.begin(115200);
    Serial2.begin(9600, SERIAL_8N1, LORA_RX, LORA_TX);
    
    StickCP2.Display.setRotation(1);
    StickCP2.Display.setTextSize(1);
    StickCP2.Display.fillScreen(BLACK);
    
    // Display initial message
    StickCP2.Display.setTextColor(GREEN);
    StickCP2.Display.drawString("LoRa Sender", 10, 10);
    StickCP2.Display.drawString("Address: 0002", 10, 30);
    StickCP2.Display.setTextColor(WHITE);
    
    Serial.println("\n=== LoRa Sender 0002 ===");
    
    delay(2000);
    
    // Configure LoRa Module
    configureLoRa();
    
    // Initialize DHT11
    initDHT11();
    
    // System ready
    StickCP2.Display.fillScreen(BLACK);
    StickCP2.Display.setTextColor(GREEN);
    StickCP2.Display.drawString("System Ready!", 10, 10);
    StickCP2.Display.setTextColor(WHITE);
    StickCP2.Display.drawString("Send every 10s", 10, 30);
    
    Serial.println("System ready");
    Serial.println("Sending every 10 seconds\n");
    
    delay(2000);
}

void loop() {
    M5.update();
    
    // Send data every 10 seconds
    if (millis() - lastSend >= sendInterval) {
        sendCount++;
        
        float temperature = 25.0;  // Default
        float humidity = 50.0;     // Default
        
        // Read DHT11
        if (dhtReady) {
            float h = dht.readHumidity();
            float t = dht.readTemperature();
            
            if (!isnan(h) && !isnan(t)) {
                temperature = t;
                humidity = h;
            } else {
                Serial.println("DHT read error");
            }
        }
        
        // Display on screen
        StickCP2.Display.fillScreen(BLACK);
        StickCP2.Display.setTextColor(CYAN);
        StickCP2.Display.drawString("SENDER 0002", 10, 5);
        StickCP2.Display.setTextColor(WHITE);
        
        if (!dhtReady) {
            StickCP2.Display.setTextColor(YELLOW);
            StickCP2.Display.drawString("[TEST MODE]", 10, 25);
            StickCP2.Display.setTextColor(WHITE);
        }
        
        StickCP2.Display.setTextColor(RED);
        StickCP2.Display.drawString("T: " + String(temperature, 1) + " C", 10, 45);
        StickCP2.Display.setTextColor(BLUE);
        StickCP2.Display.drawString("H: " + String(humidity, 1) + " %", 10, 65);
        StickCP2.Display.setTextColor(WHITE);
        StickCP2.Display.drawString("Sending...", 10, 85);
        
        // Send via LoRa
        String data = String(temperature, 1) + "," + String(humidity, 1);
        Serial2.println(data);
        
        // Display sent confirmation
        StickCP2.Display.setTextColor(GREEN);
        StickCP2.Display.drawString("Sent OK!", 10, 105);
        StickCP2.Display.setTextColor(WHITE);
        StickCP2.Display.drawString("Count: " + String(sendCount), 10, 120);
        
        // Serial log
        Serial.println("--- Send #" + String(sendCount) + " ---");
        Serial.println("Temp: " + String(temperature, 1) + " C");
        Serial.println("Humi: " + String(humidity, 1) + " %");
        Serial.println("Data: " + data);
        Serial.println();
        
        lastSend = millis();
    }
    
    delay(100);
}

void configureLoRa() {
    StickCP2.Display.fillScreen(BLACK);
    StickCP2.Display.drawString("LoRa Config...", 10, 10);
    
    Serial.println("Configuring LoRa...");
    
    // Setup M0, M1 pins
    pinMode(M0_PIN, OUTPUT);
    pinMode(M1_PIN, OUTPUT);
    
    // Enter configuration mode (M0=1, M1=1)
    digitalWrite(M0_PIN, HIGH);
    digitalWrite(M1_PIN, HIGH);
    delay(100);
    
    // Send configuration commands
    // Address: 0002
    Serial2.write(0xC0);  // Command header
    Serial2.write(0x00);  // Starting address
    Serial2.write(0x08);  // Length
    Serial2.write(0x00);  // ADDH
    Serial2.write(0x02);  // ADDL = 0002
    Serial2.write(0x62);  // REG0: 9600bps, 8N1, 2.4k air rate
    Serial2.write(0x00);  // REG1: 200byte packet, ambient noise on
    Serial2.write(0x00);  // REG2: Channel 0
    Serial2.write(0x44);  // REG3: RSSI on, transparent mode
    delay(100);
    
    // Return to normal mode (M0=0, M1=0)
    digitalWrite(M0_PIN, LOW);
    digitalWrite(M1_PIN, LOW);
    delay(100);
    
    StickCP2.Display.fillScreen(BLACK);
    StickCP2.Display.setTextColor(GREEN);
    StickCP2.Display.drawString("LoRa: OK", 10, 10);
    StickCP2.Display.setTextColor(WHITE);
    StickCP2.Display.drawString("Addr: 0002", 10, 30);
    StickCP2.Display.drawString("Channel: 0", 10, 50);
    
    Serial.println("LoRa configured");
    Serial.println("  Address: 0002");
    Serial.println("  Channel: 0");
    
    delay(2000);
}

void initDHT11() {
    StickCP2.Display.fillScreen(BLACK);
    StickCP2.Display.drawString("DHT11 Init...", 10, 10);
    
    dht.begin();
    delay(2000);
    
    // Test DHT11
    float h = dht.readHumidity();
    float t = dht.readTemperature();
    
    StickCP2.Display.fillScreen(BLACK);
    
    if (isnan(h) || isnan(t)) {
        StickCP2.Display.setTextColor(RED);
        StickCP2.Display.drawString("DHT11: ERROR", 10, 10);
        StickCP2.Display.setTextColor(YELLOW);
        StickCP2.Display.drawString("Test Mode", 10, 30);
        StickCP2.Display.setTextColor(WHITE);
        
        Serial.println("DHT11: Not connected");
        Serial.println("Using test data");
        dhtReady = false;
    } else {
        StickCP2.Display.setTextColor(GREEN);
        StickCP2.Display.drawString("DHT11: OK", 10, 10);
        StickCP2.Display.setTextColor(WHITE);
        StickCP2.Display.drawString("T: " + String(t, 1) + " C", 10, 30);
        StickCP2.Display.drawString("H: " + String(h, 1) + " %", 10, 50);
        
        Serial.println("DHT11: Connected");
        Serial.println("  Temp: " + String(t, 1) + " C");
        Serial.println("  Humi: " + String(h, 1) + " %");
        dhtReady = true;
    }
    
    delay(2000);
}
