#include <M5StickCPlus2.h>  //   устгасан
#include <WiFi.h>
#include <HTTPClient.h>
#include <DHT.h>

// DHT11 тохиргоо
#define DHTPIN 32         //  GPIO 0 (G0 пин)
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

// Wi-Fi тохиргоо
const char* ssid = "oita-unv-405-a";
const char* password = "oitauniversity0074";

// Ambient тохиргоо
const char* ambientHost = "ambidata.io"; // Domain ашиглах
const int ambientPort = 80;
unsigned int channelId = 97413;
const char* writeKey = "f7ec51d07d2cd15a";

// Хувьсагчдыг тодорхойлох
float temperature = 0;
float humidity = 0;
unsigned long lastUpdate = 0;
const unsigned long updateInterval = 30000; // 30 секунд

void setup() {
  M5.begin();
  M5.Lcd.setRotation(3);
  M5.Lcd.fillScreen(BLACK);
  M5.Lcd.setTextSize(2);
  M5.Lcd.setCursor(0, 0);
  M5.Lcd.println("DHT11->Ambient");
  M5.Lcd.println("Эхэллээ...");
  
  Serial.begin(115200);
  
  dht.begin();
  delay(2000);
  
  connectToWiFi();
  
  M5.Lcd.fillScreen(BLACK);
  M5.Lcd.setCursor(0, 0);
  M5.Lcd.println("Бэлэн!");
}

void loop() {
  M5.update();
  
  if (millis() - lastUpdate >= updateInterval) {
    readSensorData();
    sendToAmbient();
    displayData();
    lastUpdate = millis();
  }
  
  if (M5.BtnA.wasPressed()) {
    M5.Lcd.fillScreen(BLACK);
    M5.Lcd.setCursor(0, 0);
    M5.Lcd.println("WiFi rep...");
    connectToWiFi();
  }
  
  delay(100);
}

void connectToWiFi() {
  M5.Lcd.setCursor(0, 20);
  M5.Lcd.print("WiFi: ");
  M5.Lcd.println(ssid);
  
  WiFi.begin(ssid, password);
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {
    delay(500);
    M5.Lcd.print(".");
    Serial.print(".");
    attempts++;
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    M5.Lcd.setCursor(0, 60);
    M5.Lcd.println("Connected!");
    M5.Lcd.print("IP: ");
    M5.Lcd.println(WiFi.localIP());
    Serial.println("\nWiFi OK!");
    Serial.print("IP: ");
    Serial.println(WiFi.localIP());
  } else {
    M5.Lcd.setCursor(0, 60);
    M5.Lcd.println("WiFi error!");
    Serial.println("WiFi Failed!");
  }
  delay(2000); // ✅ Харуулах хугацаа
}

void readSensorData() {
  float newTemp = dht.readTemperature();
  float newHum = dht.readHumidity();
  
  if (!isnan(newTemp)) {
    temperature = newTemp;
  }
  if (!isnan(newHum)) {
    humidity = newHum;
  }
  
  Serial.print("Temp: ");
  Serial.print(temperature);
  Serial.print("°C, Humi: ");
  Serial.print(humidity);
  Serial.println("%");
}

void sendToAmbient() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi no connected!");
    M5.Lcd.setCursor(0, 100);
    M5.Lcd.println("WiFi error!");
    return;
  }
  
  HTTPClient http;
  
  String url = "http://" + String(ambientHost) + "/api/v2/channels/" + String(channelId) + "/data";
  
  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  
  String jsonData = "{\"writeKey\":\"" + String(writeKey) + "\"";
  jsonData += ",\"d1\":\"" + String(temperature, 1) + "\"";
  jsonData += ",\"d2\":\"" + String(humidity, 1) + "\"}";
  
  Serial.println("Илгээж байна...");
  Serial.println(url);
  Serial.println(jsonData);
  
  int httpCode = http.POST(jsonData);
  
  if (httpCode > 0) {
    Serial.print("HTTP: ");
    Serial.println(httpCode);
    Serial.println(http.getString());
    
    M5.Lcd.setCursor(0, 100);
    if (httpCode == 200) {
      M5.Lcd.setTextColor(GREEN);
      M5.Lcd.println("Sent! ✓");
    } else {
      M5.Lcd.setTextColor(YELLOW);
      M5.Lcd.print("code: ");
      M5.Lcd.println(httpCode);
    }
    M5.Lcd.setTextColor(WHITE);
  } else {
    Serial.print("error: ");
    Serial.println(httpCode);
    
    M5.Lcd.setCursor(0, 100);
    M5.Lcd.setTextColor(RED);
    M5.Lcd.print("Алдаа: ");
    M5.Lcd.println(httpCode);
    M5.Lcd.setTextColor(WHITE);
  }
  
  http.end();
}

void displayData() {
  M5.Lcd.fillRect(0, 20, 240, 80, BLACK);
  
  M5.Lcd.setCursor(0, 20);
  M5.Lcd.setTextColor(WHITE);
  M5.Lcd.setTextSize(2);
  
  M5.Lcd.print("Temp: ");
  M5.Lcd.setTextColor(RED);
  M5.Lcd.print(temperature, 1);
  M5.Lcd.setTextColor(WHITE);
  M5.Lcd.println("C");
  
  M5.Lcd.print("Humi: ");
  M5.Lcd.setTextColor(BLUE);
  M5.Lcd.print(humidity, 1);
  M5.Lcd.setTextColor(WHITE);
  M5.Lcd.println("%");
  
  M5.Lcd.setCursor(0, 80);
  M5.Lcd.setTextSize(1);
  M5.Lcd.print("IP: ");
  M5.Lcd.println(WiFi.localIP());
  
  unsigned long seconds = millis() / 1000;
  M5.Lcd.print("working: ");
  M5.Lcd.print(seconds / 60);
  M5.Lcd.println(" мин");
}
